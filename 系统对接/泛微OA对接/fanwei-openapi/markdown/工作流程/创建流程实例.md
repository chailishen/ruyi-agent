# 创建流程实例

- **文档 ID**: `276083048938733568`
- **业务模块**: 工作流程
- **是否 API 接口**: 是
- **请求 URL**: `{OpenApiOrigin}/api/workflow/core/paService/v2/doCreateRequest`
- **请求方式**: `POST`
- **Content-Type**: `application/json`

## 备注

- package com.weaver.core.demo;

import cn.hutool.http.HttpRequest;
import cn.hutool.json.JSONUtil;
import com.alibaba.fastjson.JSONArray;
import com.alibaba.fastjson.JSONObject;
import com.weaver.core.constant.TestConstant;

/**
 * 创建并加签流程
 *
 * @author:chenwh
 */
public class DoCreateRequestDemoV2 extends DemoBase {

    public static void main(String[] args) {
        String response = new DoCreateRequestDemoV2().doCreateRequest();
        System.out.println(response);
    }

    /**
     * 创建并加签流程
     * {"message":{"errcode":"0","errmsg":"流程操作成功，直接刷新流程表单","requestId":"959468107453014028","otherData":{}}}
     *
     * @return
     */
    public String doCreateRequest() {
        return doCreateRequest(TestConstant.WORKFLOW_YBLC);
    }

    public String doCreateRequest(long workflowId) {
        String url = HOST + "/api/workflow/core/paService/v2/doCreateRequest?access_token=" + TOKEN;

        JSONObject param = new JSONObject();
        param.put("userid", TestConstant.USER_CHENWH);
        param.put("workflowId", workflowId);
        param.put("requestname", "demo创建流程");
        // true：流程创建并流转失败时删除流程
        param.put("delReqFlowFaild", true);
        // 1：创建并流转流程
        param.put("isnextflow", 1);
        // true：创建并流转时，校验表单必填
        param.put("isVerifyFormRequired", true);
        // 手写表单数据
        param.put("formData", getFromDataByDataDetails());
        // 直接传递dataId
//        param.put("formData", getFromDataByDataId());

        // 加签参数
        param.put("addSignEntity", operatorIds());
//        param.put("addSignEntity", addFreeNodeData());

        return HttpRequest.post(url)
                .header("Content-Type", "application/json")
                // 参数放入body
                .body(JSONUtil.toJsonStr(param))
                .execute()
                .body();
    }

    /**
     * 创建流程时，可以直接传dataId即可。
     *
     * @return
     */
    private JSONObject getFromDataByDataId() {
        JSONObject formData = new JSONObject();
        formData.put("dataId", 889735619860865025L);
        return formData;
    }

    private JSONObject getFromDataByDataDetails() {
        JSONObject formData = new JSONObject();
        JSONArray dataDetails = new JSONArray();
        formData.put("dataDetails", dataDetails);

        // 普通字段（输入框）
        JSONObject input = new JSONObject();
        input.put("fieldId", 701893257943613457L);
        input.put("content", "多行文本");
        dataDetails.add(input);

        // 普通字段（输入框），dataKey 和 fieldId 二选一即可。
        JSONObject inputDataKey = new JSONObject();
        inputDataKey.put("dataKey", "dhwb");
        inputDataKey.put("content", "多行文本1");
//        dataDetails.add(inputDataKey);

        // 钱
        JSONObject inputMoney = new JSONObject();
        inputMoney.put("fieldId", 701893257943613111L);
        inputMoney.put("content", 999.6);
        JSONObject extendParams = new JSONObject();
        extendParams.put("moneyType", "RMB");
        inputMoney.put("extendParams", extendParams);
        dataDetails.add(inputMoney);

        // 浏览按钮
        JSONObject details = new JSONObject();
        JSONArray dataOptions = new JSONArray();
        JSONObject option = new JSONObject();
        details.put("fieldId", 701893266516770830L);
        details.put("dataOptions", dataOptions);
        option.put("content", "ydy测试d");
        option.put("optionId", 3572045400610556983L);
        dataOptions.add(option);
        dataDetails.add(details);


        // 明细数据
        // 明细第一行，某一列
        JSONObject dateIndex = new JSONObject();
        dateIndex.put("fieldId", 701893275140259847L);
        dateIndex.put("dataIndex", 1);
        dateIndex.put("content", 4);
        dataDetails.add(dateIndex);
        // 明细第一行，某一列
        JSONObject dateIndex1 = new JSONObject();
        dateIndex1.put("fieldId", 701893275140259852L);
        dateIndex1.put("dataIndex", 1);
        dateIndex1.put("content", 5);
        dataDetails.add(dateIndex1);

        // 明细行，传递dataKey 的形式，必须传递subFormId
        JSONObject dateIndex2 = new JSONObject();
        dateIndex2.put("dataKey", "mxh1");
        dateIndex2.put("dataIndex", 1);
        dateIndex2.put("content", 5);
        dateIndex2.put("subFormId", 5955300098899145213L);
//        dataDetails.add(dateIndex1);

        return formData;
    }

    /**
     * operatorIds 方式加签
     *
     * @return
     */
    private JSONObject operatorIds() {
        JSONObject param = new JSONObject();
        param.put("nextNodeEditPurview", 0);
        param.put("nodeType", 1);
        param.put("isParallel", 1);
        param.put("operatorIds", TestConstant.USER_CHENWH + "," + TestConstant.USER_ZMM);
        return param;
    }

    /**
     * addFreeNodeData 方式加签
     *
     * @return
     */
    private JSONObject addFreeNodeData() {
        JSONObject param = new JSONObject();
        JSONArray array = new JSONArray();
        param.put("addFreeNodeData", array);
        // 一个arrays1 就是一个分支
        JSONArray array1 = new JSONArray();
        JSONArray array2 = new JSONArray();
        array.add(array1);
        array.add(array2);

        // 往分支中添加节点
        JSONObject node1 = new JSONObject();
        node1.put("nodeName", "节点1");
        node1.put("operatorIds", TestConstant.USER_CHENWH);
        node1.put("ccOperators", TestConstant.USER_ZMM);
        node1.put("isModifiable", 0);
        node1.put("signType", 1);
        node1.put("ccSignType", 3);
        array1.add(node1);

        JSONObject node2 = new JSONObject();
        node2.put("nodeName", "节点2");
        node2.put("operatorIds", TestConstant.USER_YDY_D);
        node2.put("ccOperators", TestConstant.USER_ZMM);
        node2.put("isModifiable", 1);
        node2.put("signType", 2);
        node2.put("ccSignType", 4);
        array2.add(node1);
        array2.add(node2);
        return param;
    }
}

## 请求参数

| 参数名 | 必选 | 类型 | 说明 |
| --- | --- | --- | --- |
| access_token |  | String | 调用接口凭证 获取accessToken信息 |
| userid |  | Long | 用户信息 获取用户信息 |
| workflowId |  | Long | 工作流Id |
| requestname |  | String | 创建的流程实例标题（不输入按照默认标题组成生成标题） |
| remark |  | String | 签字意见内容 |
| isnextflow |  | int | 是否流转到下一节点：0 否，1 是。默认值：0 |
| requestSecLevel |  | int | 密级（系统开启密级才使用这个参数），值大于0 才生效。且开启了密级，则获取创建人能获取的最大密级。 |
| requestSecValidity |  | String | 保密期限（传递密级时才传递该参数）；非数字； |
| isVerifyFormRequired |  | boolean | isnextflow=1时，true：要校验表单必填。false：不校验表单必填。默认值：false |
| isExecuteCreateAddAction |  | boolean | true：执行创建节点前附加操作，false：不执行创建节点前附加操作。默认值：false |
| isVerifyPer |  | int | 是否验证用户创建流程权限；0：不验证。1：验证。默认值： 1； |
| requestId |  | long | 可以自定义流程ID。 |
| delReqFlowFaild |  | boolean | 创建并流转失败的流程是否要删除掉；true：删除。false：不删除。默认值：false |
| businessData |  | String | 数据来源 |
| moduleDataId |  | long | 关联模块数据ID（默认值0） |
| formData |  | json | 表单数据(详细字段说明看下方) |
| requestLevel |  | int | 紧急程度 0：正常； 1：重要； 2：紧急。默认值：0 |
| addSignEntity |  | json | 加签参数对象，若传递了则进行加签。 |
| identityId |  | long | 被代理人ID |
| identityType |  | int | 1：兼岗，2：委托 |
| selectFlowParam |  | JSON | 指定流转相关参数 |
| exceptionHandleInfo |  | map | 异常处理由用户选择操作者参数；若节点异常处理未设置由用户选择操作者，传递了也不会生效；KEY：节点ID；VALUE：异常处理操作者数据； |
| imRemind |  | boolean | isnextflow=0时，给发起人发送IM提醒 |
| emailRemind |  | boolean | isnextflow=0时，给发起人发送邮件提醒 |
| smsRemind |  | boolean | isnextflow=0时，给发起人发送短信提醒 |
| mailAndSmsMsgParam |  | json | 节点允许修改提醒方式时，流转时，可以选择是否发送短信邮件提醒 |
| isNewFormData |  | boolean | true：可以传递表单高级参数（seniorFormParam生效）。false：仅仅传递表达数据（默认值）。 |
| seniorFormParam |  | json | 表单高级参数，isNewFormData=true时才生效 |

## 返回参数

| 参数名 | 类型 | 说明 |
| --- | --- | --- |
| needDefaultValue | boolean | true：给表单赋值默认值，false：默认值 |

| userChooseMailRemind | boolean | false(默认值)：不发邮件；true：发邮件 |
